NixOS Configuration Guide

Declarative Hyprland configuration using NixOS modules and Home Manager with Flakes

Why NixOS for Hyprland?

NixOS provides a unique advantage for Hyprland ricing: your entire system configuration, including Hyprland, Waybar, themes, and all dotfiles, is declared in a single reproducible configuration. Benefits include:

  • Reproducibility: Same config = identical system on any machine
  • Rollback: Instant generation switching if something breaks
  • Declarative: Describe what you want, not how to get it
  • Home Manager: User-level config management with same benefits
  • Flakes: Modern, composable, pinned dependencies
Note: This guide assumes familiarity with NixOS basics. If new to NixOS, read the official manual first.

Flakes-Based Configuration

We'll use the modern Flakes approach for a clean, maintainable setup.

Directory Structure

📁 nixos-config/ ├── flake.nix ├── hosts/ │ └── 📁 your-hostname/ │ ├── configuration.nix │ └── hardware-configuration.nix ├── modules/ │ ├── hyprland.nix │ ├── waybar.nix │ ├── graphics.nix │ └── desktop.nix └── home/ ├── home.nix ├── hyprland.nix ├── waybar.nix └── programs.nix

flake.nix

{
  description = "Hyprland NixOS Configuration";

  inputs = {
    nixpkgs.url = "github:NixOS/nixpkgs/nixos-unstable";
    home-manager = {
      url = "github:nix-community/home-manager/release-24.05";
      inputs.nixpkgs.follows = "nixpkgs";
    };
    hyprland = {
      url = "github:hyprwm/Hyprland";
      inputs.nixpkgs.follows = "nixpkgs";
    };
    nixpkgs-unstable.url = "github:NixOS/nixpkgs/nixos-unstable";
  };

  outputs = { self, nixpkgs, home-manager, hyprland, nixpkgs-unstable, ... }@inputs: {
    nixosConfigurations."your-hostname" = nixpkgs.lib.nixosSystem {
      system = "x86_64-linux";
      modules = [
        ({ pkgs, ... }: {
          imports = [ ./modules/hyprland.nix ./modules/waybar.nix ./modules/graphics.nix ./modules/desktop.nix ];
        })
        home-manager.nixosModules.home-manager
        {
          home-manager.useGlobalPkgs = true;
          home-manager.useUserPackages = true;
          home-manager.users.your-user = import ./home/home.nix;
        }
        ./hosts/your-hostname/configuration.nix
      ];
    };
  };
}

Hyprland NixOS Module

Create modules/hyprland.nix for system-level Hyprland configuration:

{ config, pkgs, lib, ... }:

{
  # Enable Hyprland
  programs.hyprland = {
    enable = true;
    package = pkgs.hyprland;  # or pkgs.hyprland-git for latest

    # Optional: enable plugins
    plugins = with pkgs; [
      hyprland-plugins  # from hyprland-contrib
    ];
  };

  # Wayland environment
  hardware.opengl.enable = true;

  # D-Bus for desktop integration
  services.dbus.packages = with pkgs; [ dbus ];

  # Polkit for GUI authentication
  services.polkit.enable = true;

  # XDG Desktop Portal for screen sharing, file picking
  xdg.portal = {
    enable = true;
    extraPortals = with pkgs; [ xdg-desktop-portal-hyprland xdg-desktop-portal-gtk ];
  };

  # Fonts
  fonts.fonts = with pkgs; [
    jetbrains-mono-ttf
    noto-fonts
    noto-fonts-emoji
    noto-fonts-cjk
    nerd-fonts
  ];

  # Environment variables for Wayland
  environment.sessionVariables = {
    XDG_CURRENT_DESKTOP = "Hyprland";
    XDG_SESSION_DESKTOP = "Hyprland";
    QT_QPA_PLATFORM = "wayland";
    QT_WAYLAND_DISABLE_WINDOWDECORATION = "1";
    GDK_BACKEND = "wayland,x11";
    CLUTTER_BACKEND = "wayland";
    SDL_VIDEODRIVER = "wayland";
    _JAVA_AWT_WM_NONREPARENTING = "1";
    MOZ_ENABLE_WAYLAND = "1";
  };
}

Hardware Graphics (modules/graphics.nix)

{ config, pkgs, ... }:

{
  hardware.opengl = {
    enable = true;
    driSupport = true;
    driSupport32Bit = true;
  };

  # AMD specific
  hardware.enableRedistributableFirmware = true;

  # For newer AMD GPUs
  services.udev.packages = [ pkgs.linux-firmware ];
}
{ config, pkgs, ... }:

{
  hardware.opengl = {
    enable = true;
    driSupport = true;
    driSupport32Bit = true;
  };

  # Intel specific - enable VA-API
  hardware.video.hardwareAcceleration = {
    enable = true;
    intel = true;
  };
}
{ config, pkgs, ... }:

{
  # NVIDIA proprietary drivers
  services.xserver.videoDrivers = [ "nvidia" ];

  hardware.nvidia = {
    modesetting.enable = true;
    powerManagement.enable = true;
    powerManagement.finegrained = true;
    nvidiaSettings = true;
    nvidiaPersistenced = true;
    package = config.boot.kernelPackages.nvidiaPackages.stable;

    # For optimus laptops
    optimus.prime = {
      enable = true;
      intelBusId = "PCI:0:2:0";
      nvidiaBusId = "PCI:1:0:0";
    };
  };

  # Kernel parameters
  boot.kernelParams = [ "nvidia_drm.modeset=1" "nvidia_drm.fbdev=1" ];

  # Environment variables for Wayland
  environment.sessionVariables = {
    LIBVA_DRIVER_NAME = "nvidia";
    __GLX_VENDOR_LIBRARY_NAME = "nvidia";
    WLR_NO_HARDWARE_CURSORS = "1";
    GBM_BACKEND = "nvidia-drm";
    __GL_GSYNC_ALLOWED = "0";
    __GL_VRR_ALLOWED = "0";
  };
}

Home Manager Configuration

User-level configuration in home/home.nix:

{ config, pkgs, lib, ... }:

let
  # Import our modules
  hyprlandConfig = import ./hyprland.nix { inherit config pkgs lib; };
  waybarConfig = import ./waybar.nix { inherit config pkgs lib; };
  programsConfig = import ./programs.nix { inherit config pkgs lib; };
in
{
  imports = [ hyprlandConfig waybarConfig programsConfig ];

  home.username = "your-user";
  home.homeDirectory = "/home/your-user";

  # User packages
  home.packages = with pkgs; [
    hyprland
    waybar
    wofi
    swaybg
    swaylock-effects
    grim
    slurp
    swappy
    cliphist
    wl-clipboard
    swaync
    brightnessctl
    playerctl
    pamixer
    kitty
    thunar
    btop
    neovim
    git
    firefox
    jetbrains-mono-ttf
    nerd-fonts
  ];

  # GTK theme
  xsettings.gtk = {
    enable = true;
    themeName = "catppuccin-gtk-theme-mocha";
    iconThemeName = "Papirus-Dark";
    cursorThemeName = "Bibata-Modern-Classic";
    fontName = "JetBrains Mono 11";
  };

  # Fontconfig
  fontconfig = {
    enable = true;
    userFontDirectories = [ ./fonts ];
  };

  # Shell
  programs.zsh = {
    enable = true;
    oh-my-zsh = {
      enable = true;
      theme = "agnoster";
      plugins = [ "git" "nix-shell" "command-not-found" ];
    };
  };

  # Git config
  programs.git = {
    enable = true;
    userName = "Your Name";
    userEmail = "your@email.com";
    signing.key = "YOUR_GPG_KEY";
  };
}

GPU-Specific Configuration

Refer to the modules/graphics.nix examples in the Hyprland module section above for AMD, Intel, and NVIDIA configurations.

Hybrid Graphics (Optimus)

For laptops with both integrated and discrete GPUs:

{ config, pkgs, ... }:

{
  hardware.nvidia = {
    optimus = {
      enable = true;
      # Switch between GPUs using:
      # __NV_PRIME_RENDER_OFFLOAD=1 __GLX_VENDOR_LIBRARY_NAME=nvidia 
    };
  };

  # Add to hyprland keybindings for GPU switching
  programs.hyprland.settings.binds = [
    "bind = SUPER, G, exec, __NV_PRIME_RENDER_OFFLOAD=1 __GLX_VENDOR_LIBRARY_NAME=nvidia kitty"
  ];
}

Updating Your System

Update Flake Inputs

# Update all inputs
nix flake update

# Update specific input
nix flake lock --update-input nixpkgs
nix flake lock --update-input hyprland
nix flake lock --update-input home-manager

Rebuild System

# Rebuild and switch
sudo nixos-rebuild switch --flake .#your-hostname

# Or with verbose output
sudo nixos-rebuild switch --flake .#your-hostname --show-trace

# Dry run to see what would change
sudo nixos-rebuild dry-build --flake .#your-hostname

Rollback

# List generations
sudo nixos-rebuild list-generations

# Switch to previous generation
sudo nixos-rebuild switch --flake .#your-hostname --generation 42

# Or from boot menu (GRUB/systemd-boot)

Home Manager Updates

# Update home manager only
home-manager switch --flake .#your-user@your-hostname

# Or as part of system rebuild (automatic with nixos-rebuild)

Common NixOS Issues

Hyprland not in PATH

# Ensure hyprland is in home.packages or programs.hyprland.enable = true
home.packages = with pkgs; [ hyprland ];  # or hyprland-git

Waybar not starting

# Check waybar config syntax
waybar -c ~/.config/waybar/config.jsonc -s ~/.config/waybar/style.css

# Check logs
journalctl --user -u waybar -f

Font issues

# Rebuild font cache
fc-cache -fv

# Check available fonts
fc-list :family | sort -u

Permission denied on rebuild

# Fix store permissions
sudo nix-store --verify --check-contents --repair

# Or garbage collect
sudo nix-collect-garbage -d

Flake evaluation errors

# Debug flake
nix flake check --show-trace

# Check specific output
nix eval .#nixosConfigurations.your-hostname.config.system.build.toplevel --show-trace
